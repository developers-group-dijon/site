{{- /* Ouvre dans un nouvel onglet les liens externes (http et https) du contenu Asciidoc, sans JavaScript : ajoute target="_blank" rel="noopener" à chaque lien absolu, sauf vers le site lui-même. L'ordre des attributs varie (class peut précéder href), d'où les [^>]*? dans les motifs. Un window=_blank écrit dans l'article n'est donc pas nécessaire. Usage : {{ partial "liens-externes.html" .Content }} */ -}} {{- $cible := ` target="_blank" rel="noopener"` -}} {{- $html := string . -}} {{- /* Retirer d'abord un éventuel window=_blank (Hugo peut réordonner target et rel) */ -}} {{- $html = replaceRE `(]*?) target="_blank"` `${1}` $html -}} {{- $html = replaceRE `(]*?) rel="noopener"` `${1}` $html -}} {{- /* Ajout après le href de tout lien absolu */ -}} {{- $html = replaceRE `(]*? href="https?://[^"]+")` (printf "${1}%s" $cible) $html -}} {{- /* Liens absolus vers le site lui-même : même onglet */ -}} {{- $html = replaceRE (printf `(]*? href="https?://(?:www\.)?developers-group-dijon\.fr(?:[/?#][^"]*)?")([^>]*?)%s` $cible) `${1}${2}` $html -}} {{- $html | safeHTML -}}
